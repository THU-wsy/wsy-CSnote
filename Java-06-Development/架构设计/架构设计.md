# 扩展点拆分

## 扩展点工厂

**扩展功能注解**：使用该注解标注的接口代表一类功能扩展点

```java
@Documented
@Retention(RetentionPolicy.RUNTIME)
@Target({ElementType.TYPE})
public @interface ProcessAbility {
}
```

**扩展点路由注解**：使用该注解标注的类代表一个具体的扩展点实现

```java
@Documented
@Retention(RetentionPolicy.RUNTIME)
@Target({ElementType.TYPE})
@Component
public @interface ProcessAbilityIndicator {
    
    // 流程标识维度
    ProcessIdentityIndicator processIdentityIndicator() default @ProcessIdentityIndicator;
    
    // 其余自定义扩展维度，支持五个维度
    String abilityAnchor1() default ProcessAbilityFactory.DEFAULT_ANCHOR_1;
    String abilityAnchor2() default ProcessAbilityFactory.DEFAULT_ANCHOR_2;
    String abilityAnchor3() default ProcessAbilityFactory.DEFAULT_ANCHOR_3;
    String abilityAnchor4() default ProcessAbilityFactory.DEFAULT_ANCHOR_4;
    String abilityAnchor5() default ProcessAbilityFactory.DEFAULT_ANCHOR_5;
}
```

**扩展点路由组注解**：可以管理多个扩展点路由注解

```java
@Documented
@Retention(RetentionPolicy.RUNTIME)
@Target({ElementType.TYPE})
@Component
public @interface ProcessAbilityIndicatorGroup {
    
    ProcessAbilityIndicator[] group();
}
```

**默认扩展点路由注解**：

```java
@Documented
@Retention(RetentionPolicy.RUNTIME)
@Target({ElementType.TYPE})
@Component
public @interface DefaultProcessAbilityIndicator {
}
```

**流程标识相关类**：

```java
// 流程标识定位注解
@Documented
@Retention(RetentionPolicy.RUNTIME)
@Target({ElementType.TYPE})
@Component
public @interface ProcessIdentityIndicator {

    InvestmentModeEnum investmentMode() default InvestmentModeEnum.DEFAULT;

    EntranceCodeEnum entranceCode() default EntranceCodeEnum.DEFAULT;
}

// 流程标识
@Data
public class ProcessIdentity implements Serializable {
    
    private InvestmentModeEnum investmentMode;
    
    private EntranceCodeEnum entranceCode;
}

// 相关业务枚举
@AllArgsConstructor
@Getter
public enum InvestmentModeEnum {

    DEFAULT("DEFAULT", "默认"),
    DZ_BUSINESS("DZ_BUSINESS", "到综"),
    DC_BUSINESS("DC_BUSINESS", "到餐");

    private final String code;
    private final String desc;
    
    public static InvestmentModeEnum getEnumByCode(String code) {
        for (InvestmentModeEnum investmentModeEnum : InvestmentModeEnum.values()) {
            if (investmentModeEnum.getCode().equals(code)) {
                return investmentModeEnum;
            }
        }
        return null;
    }
}

@AllArgsConstructor
@Getter
public enum EntranceCodeEnum {

    DEFAULT("DEFAULT", "默认"),
    PC("PC", "PC端"),
    APP("APP", "APP端");

    private final String code;
    private final String desc;

    public static EntranceCodeEnum getEnumByCode(String code) {
        for (EntranceCodeEnum entranceCodeEnum : EntranceCodeEnum.values()) {
            if (entranceCodeEnum.getCode().equals(code)) {
                return entranceCodeEnum;
            }
        }
        return null;
    }
}
```

**扩展点工厂（核心）**：

```java
@Component
@Slf4j
public class ProcessAbilityFactory implements InitializingBean, ApplicationContextAware {

    public static final String DELIMITER = "-";
    public static final String DEFAULT_ANCHOR_1 = "__ANCHOR1__";
    public static final String DEFAULT_ANCHOR_2 = "__ANCHOR2__";
    public static final String DEFAULT_ANCHOR_3 = "__ANCHOR3__";
    public static final String DEFAULT_ANCHOR_4 = "__ANCHOR4__";
    public static final String DEFAULT_ANCHOR_5 = "__ANCHOR5__";

    private final Map<Class<?>, Map<String, Object>> abilityMap = new ConcurrentHashMap<>();
    private final Map<Class<?>, Object> defaultAbilityMap = new ConcurrentHashMap<>();

    private ApplicationContext applicationContext;

    @Override
    public void setApplicationContext(ApplicationContext applicationContext) throws BeansException {
        this.applicationContext = applicationContext;
    }

    @Override
    public void afterPropertiesSet() throws Exception {
        try {
            // 从Spring容器中获取所有标注了@ProcessAbility注解的Bean
            Map<String, Object> beansWithAnnotation = applicationContext.getBeansWithAnnotation(ProcessAbility.class);
            for (Object ability : beansWithAnnotation.values()) {
                Class<?> abilityClass = AopUtils.getTargetClass(ability);
                registerAbility(abilityClass, ability);
            }
            log.info("ProcessAbilityFactory初始化成功，Ability数量为{}", beansWithAnnotation.size());
        } catch (Exception e) {
            log.error("ProcessAbilityFactory初始化失败", e);
            throw new RuntimeException("ProcessAbilityFactory初始化失败", e);
        }
    }

    /**
     * 注册Ability实例
     */
    private void registerAbility(Class<?> abilityClass, Object ability) {
        // 处理默认Indicator
        DefaultProcessAbilityIndicator defaultAnnotation = abilityClass.getAnnotation(DefaultProcessAbilityIndicator.class);
        if (defaultAnnotation != null) {
            doRegisterDefaultIndicator(abilityClass, ability);
        }

        // 处理单个Indicator
        ProcessAbilityIndicator singleAnnotation = abilityClass.getAnnotation(ProcessAbilityIndicator.class);
        if (singleAnnotation != null) {
            doRegisterSingleIndicator(singleAnnotation, abilityClass, ability);
        }

        // 处理IndicatorGroup
        ProcessAbilityIndicatorGroup groupAnnotation = abilityClass.getAnnotation(ProcessAbilityIndicatorGroup.class);
        if (groupAnnotation != null && groupAnnotation.group() != null) {
            for (ProcessAbilityIndicator indicator : groupAnnotation.group()) {
                doRegisterSingleIndicator(indicator, abilityClass, ability);
            }
        }
    }

    private Class<?> generateProcessAbilityInterface(Class<?> abilityClass) {
        Class<?>[] interfaces = abilityClass.getInterfaces();
        if (interfaces.length != 1) {
            throw new RuntimeException(abilityClass.getSimpleName() + "未实现接口或实现了多个接口，无法注册");
        }
        return interfaces[0];
    }

    private String generateIndicatorKey(ProcessAbilityIndicator indicator) {
        StringJoiner stringJoiner = new StringJoiner(DELIMITER);

        ProcessIdentityIndicator processIdentityIndicator = indicator.processIdentityIndicator();
        stringJoiner.add(processIdentityIndicator.investmentMode().getCode());
        stringJoiner.add(processIdentityIndicator.entranceCode().getCode());

        stringJoiner.add(indicator.abilityAnchor1());
        stringJoiner.add(indicator.abilityAnchor2());
        stringJoiner.add(indicator.abilityAnchor3());
        stringJoiner.add(indicator.abilityAnchor4());
        stringJoiner.add(indicator.abilityAnchor5());

        return stringJoiner.toString();
    }

    private void doRegisterDefaultIndicator(Class<?> abilityClass, Object ability) {
        Class<?> classKey = generateProcessAbilityInterface(abilityClass);
        if (defaultAbilityMap.containsKey(classKey)) {
            throw new RuntimeException("defaultAbilityMap中发生重复注册==>" + classKey.getSimpleName());
        }
        defaultAbilityMap.put(classKey, ability);
    }

    private void doRegisterSingleIndicator(ProcessAbilityIndicator indicator, Class<?> abilityClass, Object ability) {
        Class<?> classKey = generateProcessAbilityInterface(abilityClass);
        Map<String, Object> map = abilityMap.computeIfAbsent(classKey, k -> new ConcurrentHashMap<>());
        String indicatorKey = generateIndicatorKey(indicator);
        if (map.containsKey(indicatorKey)) {
            throw new RuntimeException("abilityMap中发生重复注册==>classKey:" + classKey.getSimpleName() + ", indicatorKey:" + indicatorKey);
        }
        map.put(indicatorKey, ability);
    }

    /**
     * 核心获取Ability方法
     */
    public <T> T getAbility(Class<T> tClass, ProcessIdentity processIdentity, String ... anchors) {
        if (tClass == null) {
            return null;
        }
        Map<String, Object> map = abilityMap.get(tClass);
        if (map != null) {
            Object ability = map.get(generateMatchKey(processIdentity, anchors));
            if (tClass.isInstance(ability)) {
                return tClass.cast(ability);
            }
        }
        Object defaultAbility = defaultAbilityMap.get(tClass);
        if (tClass.isInstance(defaultAbility)) {
            return tClass.cast(defaultAbility);
        }
        return null;
    }

    private String generateMatchKey(ProcessIdentity processIdentity, String ... anchors) {
        String investmentMode = InvestmentModeEnum.DEFAULT.getCode();
        String entranceCode = EntranceCodeEnum.DEFAULT.getCode();
        if (processIdentity != null) {
            if (processIdentity.getInvestmentMode() != null) {
                investmentMode = processIdentity.getInvestmentMode().getCode();
            }
            if (processIdentity.getEntranceCode() != null) {
                entranceCode = processIdentity.getEntranceCode().getCode();
            }
        }

        StringJoiner stringJoiner = new StringJoiner(DELIMITER);

        stringJoiner.add(investmentMode);
        stringJoiner.add(entranceCode);

        stringJoiner.add(anchors.length >= 1 ? anchors[0] : DEFAULT_ANCHOR_1);
        stringJoiner.add(anchors.length >= 2 ? anchors[1] : DEFAULT_ANCHOR_2);
        stringJoiner.add(anchors.length >= 3 ? anchors[2] : DEFAULT_ANCHOR_3);
        stringJoiner.add(anchors.length >= 4 ? anchors[3] : DEFAULT_ANCHOR_4);
        stringJoiner.add(anchors.length >= 5 ? anchors[4] : DEFAULT_ANCHOR_5);
        return stringJoiner.toString();
    }
}
```
